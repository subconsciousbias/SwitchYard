# Development-only dependencies. Installed on top of requirements.txt for
# running tests, linting, and the pre-commit hooks. The offline test suite
# (`scripts/test.sh`) uses pytest; ruff is the project's linter; hypothesis
# is in here for property-based testing on classes that opt in.
#
# Issue #145: pytest>=8.4,<9 keeps us on a single major while letting CI pick up
# patch releases, ruff>=0.6 matches the version that ships the rules we select
# in pyproject.toml, and hypothesis>=6.100 is the minimum that supports the
# generator APIs our property-based tests rely on.
#
# litellm>=1.101.0,<2 is needed by the offline test suite: several
# tests/test_*.py modules do `from switchyard.hooks import ...` at the
# top of the file, and switchyard/hooks.py imports
# `litellm.integrations.custom_logger.CustomLogger` at module scope.
# Without it pytest fails collection with ModuleNotFoundError before a
# single test runs. The lower bound matches the gateway image's
# Dockerfile.gateway pin (ghcr.io/berriai/litellm:v1.101.0), so a test
# environment and a live gateway are talking to the same litellm
# major. Note: litellm at import time fetches its model_cost map from
# raw.githubusercontent.com; conftest.py blocks that socket per the
# "tests never call a real provider" policy, so the CI workflow and
# scripts/test.sh both export LITELLM_LOCAL_MODEL_COST_MAP=True to
# keep collection from erroring out on the network fetch.
pytest>=8.4,<9
ruff>=0.6
hypothesis>=6.100
litellm>=1.101.0,<2
# jsonschema: the sidecars validate response_format answers with it
# (Dockerfile.sidecar installs it); pinned here so the tests do not rely on
# litellm pulling it in.
jsonschema>=4.17,<5
# fakeredis[lua]: the embedded Lua-capable backend that tests/test_slots_lua.py
# uses in CI (TESTING.md "Fast run"). Without it the file's three Lua tests
# (test_scenario_matrix_lua_and_fake_match, test_lua_only_ttl_assertions,
# test_107_inflight_zset_ttl) silently SKIP — including the #107 regression
# gate. Pin>=2.30 for the modern asyncio API; lupa is the Lua runtime and
# ships as fakeredis's [lua] extra.
fakeredis[lua]>=2.30
# redis: fakeredis 2.x predates redis-py 8.x's activate_maint_notifications
# path (which subscribes via a host string fakeredis's FakeServer does not
# satisfy), and dev installs transitively pull the latest redis>=5 the moment
# nothing pins it. Pin to <6 so a fresh `pip install -r requirements-dev.txt`
# keeps the lua backend working. Production keeps the unconstrained
# redis>=5 from requirements.txt (only used by the portal + sidecar images,
# which never import fakeredis).
redis>=5.0,<6
